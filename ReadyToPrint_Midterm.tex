\question[5] Simplify: $3 - 3( 6 - |4 - 4^2| )$
\begin{solution}
$3 - 3( 6 - |4 - 4^2| )$ 

 $= 3 - 3( 6 - |4 - 16| )$ 

 $= 3 - 3( 6 - |-12| )$ 

 $= 3 - 3( 6 - 12 )$ 

 $= 3 - 3( -6 )$ 

 $= 3 - ( -18 )$ 

 $= \boxed{ 21 }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Simplify the radical expression completely: 

 $4 \sqrt[3]{ -16 }$
\begin{solution}
$4 \sqrt[3]{ -16 }$ 

 $= 4 \cdot \sqrt[3]{ -8 \cdot 2 }$ 

 $= 4 \cdot (-2) \sqrt[3]{ 2 }$ 

 $= \boxed{ -8 \sqrt[3]{ 2 } }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Simplify: $\displaystyle \left( \frac{ -4 a^{-4} b^{5} c^{-4} }{ a^{3} b^{4} c^{0} } \right)^{ -2 }$
\begin{solution}
$\displaystyle \left( \frac{ -4 a^{-4} b^{5} c^{-4} }{ a^{3} b^{4} c^{0} } \right)^{ -2 }$ 

 $= \left( -4 a^{-7} b^{1} c^{-4} \right)^{-2}$ 

 $= (-4)^{-2} a^{14} b^{-2} c^{8}$ 

 $= \boxed{ \frac{a^{14}c^{8}}{16b^{2}} }$
\end{solution}
\vspace{\stretch{1}}\answerline

\question[5] Simplify: $(4 \, x^{2} + 5) - (-2 \, x^{2} + 6 \, x + 5)$
\begin{solution}
$(4 \, x^{2} + 5) - (-2 \, x^{2} + 6 \, x + 5)$ 

 $= 4x^2+5+2x^2-6x-5$ 

 $= 4x^2+2x^2-6x+5-5$ 

 $= \boxed{ 6 \, x^{2} - 6 \, x }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Simplify: $(4x - 2y)(16x^2 + 8xy + 4y^2)$
\begin{solution}
$(4x - 2y)(16x^2 + 8xy + 4y^2)$ 

 $= 4x(16x^2 + 8xy + 4y^2) - 2y(16x^2 + 8xy + 4y^2)$ 

 $= 64x^3 + 32x^2y + 16xy^2 - 32x^2y - 16xy^2 - 8y^3$ 

 $= \boxed{ 64x^3 - 8y^3 }$
\end{solution}
\vspace{\stretch{2}}\answerline


\question[5] Simplify: $\displaystyle \frac{ -8 \, x^{4} + 20 \, x^{3} - 2 \, x^{2} - 5 \, x - 12 }{ 2 \, x - 3 }$
\begin{solution}
$\begin{array}{r} -4 \, x^{3} + 4 \, x^{2} + 5 \, x + 5 \\ 2 \, x - 3 \overline{ ) -8 \, x^{4} + 20 \, x^{3} - 2 \, x^{2} - 5 \, x - 12 } \\ \underline{ -( -8 \, x^{4} + 12 \, x^{3} ) } \\\\ 8 \, x^{3} - 2 \, x^{2} - 5 \, x - 12 \\ \underline{ -( 8 \, x^{3} - 12 \, x^{2} ) } \\\\ 10 \, x^{2} - 5 \, x - 12 \\ \underline{ -( 10 \, x^{2} - 15 \, x ) } \\\\ 10 \, x - 12 \\ \underline{ -( 10 \, x - 15 ) } \\\\ 3 \end{array}$ 

 $= \boxed{ -4 \, x^{3} + 4 \, x^{2} + 5 \, x + 5 + \frac{3}{2 \, x - 3} }$
\end{solution}
\vspace{\stretch{2}}\answerline

\question[5] Solve: $15 - 6(4x - 9) = 5(3x + 45)$
\begin{solution}
$15 - 6(4x - 9) = 5(3x + 45)$ 

 $15 - 24x + 54 = 15x + 225$ 

 $69 - 24x = 15x + 225$ 

 $69 = 39x + 225$ 

 $-156 = 39x$ 

 $\boxed{ x = -4 }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Solve: $\displaystyle \frac{1}{3}x - \frac{6}{15} = -4$
\begin{solution}
$\displaystyle \frac{1}{3}x - \frac{6}{15} = -4$ 

 $15 \left( \frac{1}{3}x - \frac{6}{15} = -4 \right)$ 

 $5x - 6 = -60$ 

 $5x = -54$ 

 $x = \frac{-54}{5}$ 

 $\boxed{ x = -\frac{54}{5} }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Solve for $y$: $dx - ey = f$
\begin{solution}
$dx - ey = f$ 

 $-ey = f - dx$ 

 $\boxed{ y = \frac{dx - f}{e} }$
\end{solution}
\vspace{\stretch{1}}\answerline

\question[5] Find the equation of the line parallel to $6x - 5y = -4$ and passing through $(-3, -5)$.
\begin{solution}
Slope:

 $6x - 5y = -4$ 

 $-5y = -6x + -4$ 

 $y = \frac{6}{5}x + \frac{4}{5}$ 

 $m = \frac{6}{5}$ 

Y-intercept:

 $y = mx + b$ 

 $-5 = \frac{6}{5}(-3) + b$ 

 $-5 = -\frac{18}{5} + b$ 

 $b = -\frac{7}{5}$ 

 $\boxed{ y = \frac{6}{5}x - \frac{7}{5} }$
\end{solution}
\vspace{\stretch{1}}\answerline

\question[5] A particular search engine assigns a ranking to a webpage based on the number of links that direct users to the webpage. The ranking of a webpage is given by the function below, where $n$ represents the number of links that direct users to it. How many links will be needed to obtain a page ranking of 10? 

 $R(n) = \frac{1}{18}n + 4$
\begin{solution}
$10 = \frac{1}{18}n + 4$ 

 $6 = \frac{1}{18}n$ 

 $108 = n$ 

 $\boxed{ 108 \text{ links} }$
\end{solution}
\vspace{\stretch{1}}\answerline


\question[5] Solve the system of equations. 

 $\begin{cases} x - y = -3 \\ x + y = 1 \end{cases}$
\begin{solution}
$x - y = -3 \to x = y - 3$ 

Substitute:

 $(y - 3) + y = 1$ 

 $2y - 3 = 1$ 

 $2y = 4$ 

 $y = 2$ 

Back-substitute:

 $x = 2 + (-3)$ 

 $x = -1$ 

 $\boxed{ (-1, 2) }$
\end{solution}
\vspace{\stretch{2}}\answerline
